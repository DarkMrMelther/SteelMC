93impl StructureBiomeProvider for FixedStructureBiomeProvider {
94    fn possible_biomes(&self) -> FxHashSet<Identifier> {
95        FxHashSet::from_iter([self.biome.key.clone()])
96    }
97
98    fn find_biome_horizontal(
99        &self,
100        origin_x: i32,
101        origin_z: i32,
102        search_radius: i32,
103        allowed: &dyn Fn(&Identifier) -> bool,
104        rng: &mut LegacyRandom,
105    ) -> Option<(i32, i32)> {
106        if !allowed(&self.biome.key) {
107            return None;
108        }
109
110        let noise_center_x = origin_x >> 2;
111        let noise_center_z = origin_z >> 2;
112        let noise_radius = search_radius >> 2;
113        let mut result = None;
114        let mut found = 0;
115        for z in -noise_radius..=noise_radius {
116            for x in -noise_radius..=noise_radius {
117                if result.is_none() || rng.next_i32_bounded(found + 1) == 0 {
118                    result = Some(((noise_center_x + x) << 2, (noise_center_z + z) << 2));
119                }
120                found += 1;
121            }
122        }
123        result
124    }
125}

240    /// Ring-placement candidates ordered by vanilla's stronghold distance pre-check.
241    #[must_use]
242    pub fn ring_candidates(&self, origin: BlockPos) -> Vec<StructureLocateCandidate> {
243        let mut candidates = Vec::new();
244        for (scan_id, placement) in self.placements.iter().enumerate() {
245            let Some(ring_positions) = &placement.ring_positions else {
246                continue;
247            };
248            for &chunk_pos in ring_positions {
249                candidates.push(StructureLocateCandidate::new_ring(
250                    scan_id,
251                    placement.placement.locate_pos(chunk_pos),
252                    chunk_pos,
253                ));
254            }
255        }
256        candidates.sort_by_key(|candidate| candidate.ring_distance_squared(origin));
257        candidates
258    }
259
260    /// Random-spread candidates on the square shell at `radius` around `origin`.
261    ///
262    /// This matches vanilla's `ChunkGenerator.getNearestGeneratedStructure` scan:
263    /// each shell step moves by the placement spacing before resolving the
264    /// potential structure chunk inside that placement cell.
265    #[must_use]
266    pub fn random_spread_candidates_at_radius(
267        &self,
268        origin: BlockPos,
269        radius: i32,
270    ) -> Vec<StructureLocateCandidate> {
271        if radius < 0 {
272            return Vec::new();
273        }
274
275        let chunk_origin_x = origin.0.x >> 4;
276        let chunk_origin_z = origin.0.z >> 4;
277        let mut candidates = Vec::new();
278
279        for (scan_id, locate_placement) in self.placements.iter().enumerate() {
280            let PlacementKind::RandomSpread {
281                spacing,
282                separation,
283                spread_type,
284            } = &locate_placement.placement.kind
285            else {
286                continue;
287            };
288
289            for x in -radius..=radius {
290                let x_edge = x == -radius || x == radius;
291                for z in -radius..=radius {
292                    let z_edge = z == -radius || z == radius;
293                    if !x_edge && !z_edge {
294                        continue;
295                    }
296
297                    let sector_x = chunk_origin_x + *spacing * x;
298                    let sector_z = chunk_origin_z + *spacing * z;
299                    let chunk_pos = StructurePlacement::get_potential_structure_chunk(
300                        self.seed,
301                        locate_placement.placement.salt,
302                        sector_x,
303                        sector_z,
304                        *spacing,
305                        *separation,
306                        *spread_type,
307                    );
308                    let candidate = StructureLocateCandidate::new(
309                        scan_id,
310                        locate_placement.placement.locate_pos(chunk_pos),
311                        chunk_pos,
312                    );
313                    candidates.push(candidate);
314                }
315            }
316        }
317
318        candidates
319    }
320}

352/// Squared block distance using vanilla's three-dimensional `BlockPos.distSqr`.
353#[must_use]
354pub fn squared_distance(a: BlockPos, b: BlockPos) -> i64 {
355    let dx = i64::from(a.0.x) - i64::from(b.0.x);
356    let dy = i64::from(a.0.y) - i64::from(b.0.y);
357    let dz = i64::from(a.0.z) - i64::from(b.0.z);
358    dx * dx + dy * dy + dz * dz
359}

361fn validate_structure_sets(structure_sets: &[(Identifier, StructureSet)]) {
362    for (set_key, set) in structure_sets {
363        assert!(
364            !set.structures.is_empty(),
365            "Structure set {set_key} must have at least one structure"
366        );
367        for (entry_index, entry) in set.structures.iter().enumerate() {
368            assert!(
369                entry.weight > 0,
370                "Structure set {set_key} entry {entry_index} has non-positive weight {}",
371                entry.weight
372            );
373        }
374        assert!(
375            !(!set.placement.frequency.is_finite()
376                || !(0.0..=1.0).contains(&set.placement.frequency)),
377            "Structure set {set_key} has invalid placement frequency {}",
378            set.placement.frequency
379        );
380        if let Some(exclusion) = &set.placement.exclusion_zone
381            && exclusion.chunk_count < 0
382        {
383            panic!(
384                "Structure set {set_key} has negative exclusion chunk_count {}",
385                exclusion.chunk_count
386            );
387        }
388
389        match &set.placement.kind {
390            PlacementKind::RandomSpread {
391                spacing,
392                separation,
393                ..
394            } => {
395                assert!(
396                    *spacing > 0,
397                    "Structure set {set_key} has non-positive spacing {spacing}"
398                );
399                assert!(
400                    *separation >= 0,
401                    "Structure set {set_key} has negative separation {separation}"
402                );
403                assert!(
404                    spacing > separation,
405                    "Structure set {set_key} has spacing {spacing} <= separation {separation}"
406                );
407            }
408            PlacementKind::ConcentricRings {
409                distance,
410                spread,
411                count,
412                ..
413            } => {
414                assert!(
415                    *distance > 0,
416                    "Structure set {set_key} has non-positive ring distance {distance}"
417                );
418                assert!(
419                    *spread > 0,
420                    "Structure set {set_key} has non-positive ring spread {spread}"
421                );
422                assert!(
423                    *count >= 0,
424                    "Structure set {set_key} has negative ring count {count}"
425                );
426            }
427        }
428    }
429}

431fn validate_structure_assets(
432    structure_sets: &[(Identifier, StructureSet)],
433    structure_data: &FxHashMap<Identifier, StructureRef>,
434    structure_impls: &FxHashMap<Identifier, Box<dyn Structure>>,
435) {
436    for (set_key, set) in structure_sets {
437        for entry in &set.structures {
438            let structure = structure_data.get(&entry.structure).unwrap_or_else(|| {
439                panic!(
440                    "Structure set {set_key} references unknown structure {}",
441                    entry.structure
442                )
443            });
444            assert!(
445                structure_impls.contains_key(&structure.structure_type),
446                "Structure set {set_key} references {} with unsupported structure type {}",
447                structure.key,
448                structure.structure_type
449            );
450        }
451    }

509    fn with_assets_for_ring_seed(
510        seed: i64,
511        ring_position_seed: i64,
512        biome_provider: &impl StructureBiomeProvider,
513        structure_sets: Vec<(Identifier, StructureSet)>,
514        assets: StructureGeneratorAssets,
515    ) -> Self {
516        validate_structure_sets(&structure_sets);
517
518        let structure_data: FxHashMap<Identifier, StructureRef> = REGISTRY
519            .structures
520            .iter()
521            .map(|(_, structure)| (structure.key.clone(), structure))
522            .collect();
523        validate_structure_assets(&structure_sets, &structure_data, &assets.structure_impls);
524
525        let possible_biomes = biome_provider.possible_biomes();
526        let structure_sets: Vec<_> = structure_sets
527            .into_iter()
528            .filter(|(_, set)| {
529                set.structures.iter().any(|entry| {
530                    structure_data
531                        .get(&entry.structure)
532                        .is_some_and(|structure| {
533                            structure.allowed_biomes.is_empty()
534                                || structure
535                                    .allowed_biomes
536                                    .iter()
537                                    .any(|biome| possible_biomes.contains(biome))
538                        })
539                })
540            })
541            .collect();
542
543        let structure_set_indices: FxHashMap<Identifier, usize> = structure_sets
544            .iter()
545            .enumerate()
546            .map(|(index, (key, _))| (key.clone(), index))
547            .collect();
548
549        let mut ring_positions = FxHashMap::default();
550        for (key, set) in &structure_sets {
551            if let PlacementKind::ConcentricRings {
552                distance,
553                spread,
554                count,
555                preferred_biomes,
556            } = &set.placement.kind
557            {
558                let mut snap =
559                    |block_x: i32, block_z: i32, rng: &mut LegacyRandom| -> Option<(i32, i32)> {
560                        biome_provider.find_biome_horizontal(
561                            block_x,
562                            block_z,
563                            112,
564                            &|biome| preferred_biomes.contains(biome),
565                            rng,
566                        )
567                    };
568                let positions = generate_ring_positions(
569                    ring_position_seed,
570                    *distance,
571                    *spread,
572                    *count,
573                    Some(&mut snap),
574                );
575                ring_positions.insert(key.clone(), positions);
576            }
577        }
578
579        Self {
580            seed,
581            structure_sets,
582            structure_set_indices,
583            structure_data,
584            ring_positions,
585            template_pools: assets.template_pools,
586            templates: assets.templates,
587            structure_impls: assets.structure_impls,
588        }
589    }

609    /// Builds a detached locate plan for one or more structure ids.
610    #[must_use]
611    pub fn locate_plan_for_structures(
612        &self,
613        structures: &[Identifier],
614    ) -> Option<StructureLocatePlan> {
615        let mut placements = Vec::new();
616        for (set_key, set) in &self.structure_sets {
617            let matching_structures = structures
618                .iter()
619                .filter(|structure| {
620                    set.structures
621                        .iter()
622                        .any(|entry| entry.structure == **structure)
623                })
624                .cloned()
625                .collect::<Vec<_>>();
626            if matching_structures.is_empty() {
627                continue;
628            }
629
630            placements.push(StructureLocatePlacement {
631                placement: set.placement.clone(),
632                ring_positions: self.ring_positions.get(set_key).cloned(),
633                structures: matching_structures,
634            });
635        }
636
637        (!placements.is_empty()).then_some(StructureLocatePlan {
638            seed: self.seed,
639            placements,
640        })
641    }

643    /// Generates structure starts for one chunk.
644    pub fn generate_starts_for_chunk(
645        &self,
646        ctx: &mut dyn StructureGenerationContext,
647        mut has_existing_start: impl FnMut(&Identifier) -> bool,
648    ) -> Vec<StructureStart> {
649        let chunk_x = ctx.chunk_x();
650        let chunk_z = ctx.chunk_z();
651        let mut generated_keys = FxHashSet::default();
652        let mut starts = Vec::new();
653
654        for (set_key, set) in &self.structure_sets {
655            if !self.is_structure_chunk_for_set(set_key, chunk_x, chunk_z, &mut Vec::new()) {
656                continue;
657            }
658
659            if set.structures.iter().any(|entry| {
660                generated_keys.contains(&entry.structure) || has_existing_start(&entry.structure)
661            }) {
662                continue;
663            }
664
665            let Some((structure, stub)) = self.select_structure(set, ctx) else {
666                continue;
667            };
668
669            let start = StructureStart::new(
670                structure.key.clone(),
671                ChunkPos::new(chunk_x, chunk_z),
672                stub.pieces,
673                structure.terrain_adjustment,
674            );
675            if !start.pieces.is_empty() {
676                generated_keys.insert(structure.key.clone());
677            }
678            starts.push(start);
679        }
680
681        starts
682    }

688    fn is_structure_chunk_for_set(
689        &self,
690        set_key: &Identifier,
691        source_x: i32,
692        source_z: i32,
693        stack: &mut Vec<Identifier>,
694    ) -> bool {
695        if stack.iter().any(|key| key == set_key) {
696            let chain = stack
697                .iter()
698                .map(ToString::to_string)
699                .chain(iter::once(set_key.to_string()))
700                .collect::<Vec<_>>()
701                .join(" -> ");
702            panic!("Circular structure exclusion zone: {chain}");
703        }
704
705        let Some(&set_index) = self.structure_set_indices.get(set_key) else {
706            return false;
707        };
708        let (_, set) = &self.structure_sets[set_index];
709        let rings = self.rings_for_set(set_key);
710        if !set
711            .placement
712            .is_structure_chunk(self.seed, source_x, source_z, rings)
713        {
714            return false;
715        }
716
717        stack.push(set_key.clone());
718        let excluded = self.is_excluded(&set.placement, source_x, source_z, stack);
719        stack.pop();
720        !excluded
721    }

723    fn is_excluded(
724        &self,
725        placement: &StructurePlacement,
726        source_x: i32,
727        source_z: i32,
728        stack: &mut Vec<Identifier>,
729    ) -> bool {
730        let Some(exclusion) = &placement.exclusion_zone else {
731            return false;
732        };
733
734        for dx in (source_x - exclusion.chunk_count)..=(source_x + exclusion.chunk_count) {
735            for dz in (source_z - exclusion.chunk_count)..=(source_z + exclusion.chunk_count) {
736                if self.is_structure_chunk_for_set(&exclusion.other_set, dx, dz, stack) {
737                    return true;
738                }
739            }
740        }
741        false
742    }
743
744    fn select_structure(
745        &self,
746        set: &StructureSet,
747        ctx: &mut dyn StructureGenerationContext,
748    ) -> Option<(StructureRef, GenerationStub)> {
749        if set.structures.len() == 1 {
750            return self.try_generate_entry(&set.structures[0], ctx);
751        }
752
753        let mut rng = LegacyRandom::from_seed(0);
754        rng.set_large_feature_seed(self.seed, ctx.chunk_x(), ctx.chunk_z());
755
756        let mut remaining: Vec<&StructureSelectionEntry> = set.structures.iter().collect();
757        let mut total_weight: i32 = remaining.iter().map(|entry| entry.weight).sum();
758
759        while !remaining.is_empty() {
760            let mut choice = rng.next_i32_bounded(total_weight);
761            let mut selected_idx = 0;
762            for (idx, entry) in remaining.iter().enumerate() {
763                choice -= entry.weight;
764                if choice < 0 {
765                    selected_idx = idx;
766                    break;
767                }
768            }
769
770            let candidate = remaining[selected_idx];
771            if let Some(generated) = self.try_generate_entry(candidate, ctx) {
772                return Some(generated);
773            }
774
775            total_weight -= candidate.weight;
776            remaining.remove(selected_idx);
777        }
778
779        None
780    }
781
782    fn try_generate_entry(
783        &self,
784        entry: &StructureSelectionEntry,
785        ctx: &mut dyn StructureGenerationContext,
786    ) -> Option<(StructureRef, GenerationStub)> {
787        let Some(structure) = self.structure_data.get(&entry.structure).copied() else {
788            tracing::warn!("Missing structure registry data for {}", entry.structure);
789            return None;
790        };
791
792        if let Some(structure_impl) = self.structure_impls.get(&structure.structure_type) {
793            let mut rng = LegacyRandom::from_seed(0);
794            rng.set_large_feature_seed(self.seed, ctx.chunk_x(), ctx.chunk_z());
795            return structure_impl
796                .find_generation_point(ctx, structure, &mut rng)
797                .map(|stub| (structure, stub));
798        }
799
800        tracing::warn!(
801            "Unknown structure type {:?} for {}, skipping structure start",
802            structure.structure_type,
803            structure.key
804        );
805        None
806    }
807}
